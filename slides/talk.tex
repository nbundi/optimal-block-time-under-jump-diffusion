\section{The case for faster blocks}
% =====================================================================

\begin{frame}{Two live conversations about how fast a chain should be}
  \begin{rwexample}[Arbitrum Research --- ``The power of faster blocks'' (Felten, May 2024)]
    ``Arbitrum's $250$ millisecond block time leads to \hl{$65\%$ lower
    arbitrage loss} compared to a $2$ second block time.''
    \blockmarker{\weblink{https://research.arbitrum.io/t/the-power-of-faster-blocks/9609}%
      {research.arbitrum.io/t/the-power-of-faster-blocks/9609}}
  \end{rwexample}
  \begin{rwexample}[EIP-7782 ``Reduce Block Latency'' --- $12$ s $\to$ $6$ s, drafted for Glamsterdam]
    ``More frequent blocks \hl{decrease LVR} (Loss Versus Rebalancing), which
    improves the economics for liquidity providers. More liquidity means lower
    spreads.''
    \blockmarker{\weblink{https://eips.ethereum.org/EIPS/eip-7782}%
      {eips.ethereum.org/EIPS/eip-7782}}
  \end{rwexample}
  Both rest on the same result. Both, on the numbers below, overstate the gain
  --- by roughly a factor of two.
\end{frame}

\begin{frame}{Where that argument comes from}
  Passive liquidity providers on an AMM lose to arbitrageurs whenever the
  external price moves --- \hl{loss-versus-rebalancing} (LVR), Milionis et
  al.\ (2022).
  \begin{itemize}
    \item Under geometric Brownian motion, a constant-product pool loses at the
          rate $\sigma^2 V/8$, independent of the block time.
    \item Add a swap fee $\gamma$ and Poisson blocks of mean $\dt$
          (Milionis et al., 2023): the rate is multiplied by
          $F(\kappa)$, $\kappa = \gamma/(\sigma\sqrt{\dt})$ ---
          the stationary probability that an arriving block offers a
          \emph{profitable} trade.
    \item $F(\kappa) \sim \sigma\sqrt{\dt}/(\sqrt2\gamma) \to 0$ as $\dt\to0$.
          That $\sqrt{\dt}$ is where Arbitrum's $65\%$ comes from, and the
          ${\sim}29\%$ implied by halving Ethereum's slot; and it says shorter
          blocks mean less LVR, all the way down.
  \end{itemize}
  \begin{takeaway}[The deployed prescription]
    ``Shorten blocks.'' Solana at $400$~ms, Ethereum L2s at $2$~s, app-chains
    chasing $50$~ms --- with LP welfare among the stated reasons.
  \end{takeaway}
\end{frame}

\begin{frame}{The primitive that prescription assumes}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      The whole argument rests on the reference price being \hl{continuous}.
      A diffusive gap has size $O(\sigma\sqrt{\dt})$, so halving the block
      halves what the arbitrageur finds.
      \medskip

      Crypto prices are not continuous:
      \begin{itemize}
        \item Scaillet et al.\ (2020): frequent, clustered jumps in BTC at high
              frequency.
        \item Saef et al.\ (2024): BTC jumps on ${\sim}58\%$, ETH on
              ${\sim}32\%$ of trading days.
        \item My own ETH/USDT calibration: $\lambda = 283$ jumps/yr.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{pitfall}[A jump does not care about $\dt$]
        A jump of size $J$ opens a price gap of size $J$ whatever the block
        schedule. Faster blocks clear each gap \emph{sooner} --- they do not
        make it \emph{smaller}.
      \end{pitfall}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contributions}
  \begin{enumerate}
    \item An \hl{additive LVR decomposition} under any finite-activity jump
          law: the classical diffusion term $\sigma^2V/8$ plus a jump term
          carrying no $\dt$ (Theorem~1).
    \item A \hl{separated rate} with fees and discrete blocks,
          $\ell_0 = \tfrac{\sigma^2V}{8}F(\kappa) + \lambda V G_\nu(\gamma)$ ---
          the two channels meeting only in a remainder bounded on both sides,
          with no order symbols (Theorem~2, Proposition~1).
    \item An \hl{exact jump floor}: $\ell(\dt) \ge \lambda V G_\nu(\gamma) > 0$
          at every $\dt$ for symmetric $\nu$, with no remainder, and approached
          only as $\sqrt{\dt}$ (Theorem~3, Corollary~1).
    \item A \hl{closed-form planner's optimum} --- a cubic, invariant in pool
          size exactly and in $(\lambda,m,\delta)$ below the flatness of the
          objective (Theorem~4), calibrated to ETH/USDT, 2020--2026.
  \end{enumerate}
  \begin{takeaway}[Headline]
    The block schedule governs one channel and cannot touch the other. At
    Ethereum's $12$~s that is $471\bpyr$ against an exact floor of $125$, and
    an LP-side optimum near $8$~s --- so the two proposals above are worth
    \hl{$35\%$} and \hl{$14\%$}, not $65\%$ and $29\%$.
  \end{takeaway}
\end{frame}

\section{Two channels, and a floor}
% =====================================================================

\begin{frame}{Setup}
  Milionis et al.\ with \hl{one primitive changed}: the reference price jumps.
  \[
    \frac{dS_t}{S_{t-}} \;=\; \mu\,dt + \sigma\,dW_t + (e^{J_t}-1)\,dN_t,
    \qquad N \text{ Poisson}(\lambda),\quad J \sim \nu .
  \]
  \begin{itemize}
    \item $\nu$ finite activity with $\int (e^{j/2}-1)^2 \nu(dj) < \infty$;
          Merton $J\sim\mathcal N(m,\delta^2)$ the running instance.
    \item CPMM, $x_ty_t = L$, pool value $V(P) = 2\sqrt{LP}$ --- concave, and
          that concavity \emph{is} the loss.
    \item Fee $\gamma$ in log-price units $\Rightarrow$ no-arbitrage band
          $\{|\log S - \log P| \le \gamma\}$; blocks Poisson of rate $1/\dt$.
    \item $\mathrm{LVR}(T) = \int_0^T V'(P_{s-})\,dP_s - [V(P_T)-V(P_0)]$;
          rate $\ell = \E_t[d\mathrm{LVR}_t]/dt$.
  \end{itemize}
  \begin{block}{What is general and what is not}
    Results need \hl{symmetry} of $\nu$, not Gaussianity. Theorems 1, 2, 4 hold
    for arbitrary $\nu$; Proposition~1 and Theorem~3 need symmetry.
  \end{block}
\end{frame}

\begin{frame}{Theorem 1 --- the zero-fee decomposition}
  In the continuous-arbitrage limit ($\gamma=0$, $\dt\to0$), $P_t \equiv S_t$
  and LVR is the It\^o--L\'evy remainder of $V(S_t)$:
  \[
    \ell \;=\;
      \underbrace{\frac{\sigma^2}{8}V(S_t)}_{\text{diffusion-LVR}}
      \;+\;
      \underbrace{\frac{\lambda}{2}V(S_t)\,\E\big[(e^{J/2}-1)^2\big]}_{\text{jump-LVR}} ,
  \]
  and for Merton $\E[(e^{J/2}-1)^2] = e^{m+\delta^2/2} - 2e^{m/2+\delta^2/8} + 1
  \ge 0$, zero iff $J \equiv 0$.
  \medskip

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Two observations}
        The jump term carries \hl{no $\dt$}. At $\gamma=0$ the arbitrageur
        clears any gap in the next block, so refining the schedule only
        redistributes \emph{when} the loss lands.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Measure}
        The drift cancels between the two terms, so no normalisation of $\mu$
        is needed --- but with jumps the rate is \emph{not} measure-independent,
        so this is a physical-measure statement and calibration is against
        realised prices.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The mispricing process}
  Theorem~1 had $\gamma=0$, so the pool tracked the reference exactly. Switch
  the fee on and a \hl{state} appears between them: $z_t := \log(S_t/P_t)$.
  \begin{itemize}
    \item Between blocks $P$ is frozen, so $z$ inherits the dynamics of
          $\log S$ --- diffusion \emph{and} jumps.
    \item At a block the arbitrageur trades iff $|z| > \gamma$, and moves the
          pool to the \emph{near band edge}: $z \mapsto
          \bound(z,-\gamma,\gamma)$ --- \hl{not to $0$}. The mispricing
          persists.
  \end{itemize}
  \[
    z_t \;=\; \big(\mu-\tfrac12\sigma^2\big)t + \sigma W_t
    + \underbrace{\textstyle\sum_{k:\,T_k\le t} J_k}_{\text{exogenous, rate }\lambda}
    + \underbrace{\textstyle\sum_{i:\,\tau_i\le t}\mathcal{J}_i}_{\text{endogenous, rate }1/\dt}
  \]
  \begin{takeaway}[The asymmetry every result comes from]
    \emph{Two} point processes drive $z$: price jumps push it \hl{away} from
    the band at a rate the protocol does not choose; block-triggered resets
    push it \hl{back} at a rate the protocol does choose.
  \end{takeaway}
\end{frame}

to $\pm\gamma$ --- teal a diffusive exceedance, orange a jump.]%
{The mispricing process, drawn}{fig_mispricing_path}

\begin{frame}{From the process to a rate}
  A block with pre-trade mispricing $z$ moves the pool's log-price by
  $\xi = \mathrm{sign}(z)(|z|-\gamma)_+$, and by the computation behind
  Theorem~1 the LP's loss is $h(\xi) = \tfrac{V}{2}(e^{\xi/2}-1)^2$. Blocks
  are Poisson and independent of $z$, so the pre-block law is the stationary
  law $\pi$ and
  \[
    \ell(\dt) \;=\; \frac{1}{\dt}\,\E_\pi\big[h(\xi)\big],
    \qquad
    h(\xi) \;=\; \underbrace{\tfrac{V}{8}\xi^2}_{\text{quadratic surrogate}}
    \;+\; \tfrac{V}{2}\,\rho(\xi),
    \quad \rho(\xi) := (e^{\xi/2}-1)^2 - \tfrac{\xi^2}{4}.
  \]
  Everything below is read off this one expression, and two exact facts about
  the split carry it:
  \begin{itemize}
    \item $\tfrac{V}{8}\xi^2$ is \hl{convex} in $z$, which $h$ itself is not
          --- a downward move of any size costs at most $V/2$. Convexity is what
          the floor (Theorem~3) needs.
    \item $\rho$ is \hl{signed} under symmetry:
          $\tfrac12[\rho(\xi)+\rho(-\xi)] = \sum_{n\ge2}
          \tfrac{1-2^{1-2n}}{(2n)!}\xi^{2n} \ge 0$, leading term
          $\tfrac{7}{192}\xi^4$. So the surrogate \emph{understates} the loss,
          and the residual is carried exactly, never discarded.
  \end{itemize}
\end{frame}

\begin{frame}{The two channels, separately}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Diffusion (Lemma 1).} At $\lambda=0$ the stationary law is
      uniform on the band with exponential tails, so
      \[
        \ell_{\mathrm{diff}} = \tfrac{\sigma^2}{8}V\cdot F(\kappa),
        \quad F(\kappa) = \frac{1}{1+\sqrt2\,\kappa}
      \]
      with $\kappa = \gamma/(\sigma\sqrt{\dt})$; $F(\kappa) =
      \Prob_\pi(|z|>\gamma)$, strictly decreasing, $F(\kappa)\to0$.
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Jumps (Lemma 2).} The jump contribution is $\lambda V\,
      G_\nu(\gamma)$ with
      \[
        G_\nu(\gamma) = \tfrac18\E_\nu\big[(|J|-\gamma)^2\1\{|J|>\gamma\}\big],
      \]
      a functional of $\nu$ alone. For symmetric Merton,
      $G = \tfrac{\delta^2}{8}\Psi(\gamma/\delta)$.
    \end{column}
  \end{columns}
  \medskip
  \begin{block}{The fee discount is different on each channel}
    Both channels are trimmed by the fee --- but by $F(\gamma/\sigma\sqrt{\dt})$
    on one and by $\Psi(\gamma/\delta)$ on the other. \hl{Only the first
    contains $\dt$}, because the size of a jump is set by $\delta$, not by how
    long the block lasted.
  \end{block}
\end{frame}

\begin{frame}{Theorem 2 --- the separated rate}
  \begin{block}{}
    \[
      \ell(\dt) \;=\; \ell_0(\dt) + \mathcal{E}(\dt),
      \qquad
      \ell_0(\dt) := \frac{\sigma^2}{8}V\cdot F\!\Big(\frac{\gamma}{\sigma\sqrt{\dt}}\Big)
      \;+\; \lambda V \cdot G_\nu(\gamma).
    \]
  \end{block}
  This is a \emph{definition} of $\mathcal{E}$ and carries no content until
  $\mathcal E$ is controlled.
  \medskip

  \textbf{Proposition 1} bounds it on both sides --- six signed terms, no order
  symbols, under a stated mixing condition $n_* \le 2(1+\eta^2)$ checked
  numerically over every block time, volatility and fee tier considered:
  \begin{itemize}
    \item interaction ($|M|\le\gamma$ pathwise), block clock, single-jump
          thinning, multi-jump, curvature residual (exact, closed form),
          stationary law.
    \item At the calibration: $\mathcal E(\dt)\in[-0.653,\,+0.849]\bpyr$ at
          $12$~s, never above $0.29\%$ of the rate over $50$~ms--$12$~s.
          Quadrature puts the realised remainder at $+0.165\bpyr$, inside the
          bound and of the predicted sign.
  \end{itemize}
\end{frame}

\begin{frame}{Theorem 3 --- an exact floor}
  \begin{takeaway}[No approximation, no remainder]
    Let $\nu$ be symmetric and satisfy $\E[g(\sum_1^k J)] \ge k\,\E[g(J)]$ for
    every $k$ (Merton with $m=0$ does). Then for \emph{every} $\dt>0$
    \[
      \ell(\dt)\;\ge\;\lambda V\,G_\nu(\gamma)\;=\;\inf_{\dt'>0}\ell_0(\dt')\;>\;0 .
    \]
  \end{takeaway}
  \begin{itemize}
    \item Proved from convexity of the quadratic surrogate --- it never touches
          the stationary law, which for $\lambda>0$ has no closed form. That is
          what makes it available, and why it does \emph{not} inherit
          Proposition~1's mixing condition.
    \item \textbf{Corollary 1.} $\ell_0$ is strictly increasing and the floor is
          approached only as $\sqrt{\dt}$: halving the block time removes at
          most $29\%$ of the diffusion term.
  \end{itemize}
  \begin{pitfall}[Keep two limits apart]
    $\dt\to0$ shortens the interval between blocks. The jump law is exogenous to
    the schedule, so it does \emph{not} shrink $J$.
  \end{pitfall}
\end{frame}

      \end{itemize}

\end{frame}

\begin{frame}{Theorem 4 --- netting LVR against what a block costs}
  Fixed per-block work gives a cost rate $R(\dt) = r/\dt$; a pool inherits
  $c/\dt$ with $c = r\,V/V_{\text{chain}}$. The LP-side planner minimises
  $W(\dt) = \ell_0(\dt) + c/\dt$:
  \begin{block}{}
    \[
      \eta^{\mathrm{opt}}\big(1+\eta^{\mathrm{opt}}\big)^2 = \frac{V\gamma^2}{8c},
      \qquad
      \dt^{\mathrm{opt}} = \frac{2\gamma^2}{\sigma^2(\eta^{\mathrm{opt}})^2},
    \]
  \end{block}
  a cubic with a unique positive root, closed form by Cardano.
  \begin{itemize}
    \item $\partial\dt^{\mathrm{opt}}/\partial\lambda =
          \partial\dt^{\mathrm{opt}}/\partial m =
          \partial\dt^{\mathrm{opt}}/\partial\delta = 0$: \hl{the jump
          parameters do not enter the first-order condition}.
    \item $\partial\dt^{\mathrm{opt}}/\partial V = 0$, exactly --- $V$ cancels,
          since $c \propto V$. The optimum is a property of the chain and the
          fee tier, not of the pool.
  \end{itemize}
  \begin{pitfall}[What the remainder does to it]
    That invariance is a property of $\ell_0$; $\mathcal E$ carries both
    $\lambda$ and $\dt$. Bounding it directly puts any exact minimiser in
    $[7.3,\,9.9]$~s --- the range over which $W$ is within $1\bpyr$ of its
    minimum anyway.
  \end{pitfall}
\end{frame}

\section{What this means, and what it does not}
% =====================================================================

\begin{frame}{Policy}
  \begin{rwexample}[Back to the two proposals]
    Arbitrum, $2$~s $\to$ $250$~ms: $\sqrt{\dt}$ gives $65\%$, the model gives
    $312 \to 203\bpyr$, a \hl{$35\%$} cut. EIP-7782, $12$~s $\to$ $6$~s:
    $\sqrt{\dt}$ gives $29\%$, the model gives $471 \to 403\bpyr$, a
    \hl{$14\%$} cut. Both price the diffusion channel as the whole loss.
  \end{rwexample}
  \begin{itemize}
    \item Shortening blocks can at most close $\ell(\dt) - \lambda VG$ --- and
          only as $\sqrt{\dt}$.
    \item The diffusion channel stays material into the low-millisecond range.
          \hl{So at every deployed block time, what limits block-time policy is
          not the floor --- it is the cost of consensus.}
    \item Once the floor does bind, the remaining LP-side levers are not the
          schedule:
      \begin{itemize}
        \item fee-tier design, since the binding ratio is $\gamma/\delta$;
        \item batch auctions, which remove the per-block arbitrage window at
              source;
        \item LVR-rebate designs, e.g.\ auction-managed AMMs routing
              pool-management revenue back to LPs.
      \end{itemize}
  \end{itemize}
  \begin{takeaway}
    Jumps change what block time can \emph{achieve}, not what it should
    \emph{be}.
  \end{takeaway}
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item \hl{$125\bpyr$} --- the jump floor. Exact for symmetric $\nu$, at
          every block time, with no remainder.
    \item \hl{$471\bpyr$} at Ethereum's $12$~s, $73\%$ of it in the channel the
          schedule governs --- and that channel yields only as $\sqrt{\dt}$.
    \item \hl{$8.4$~s} --- the LP-side optimum. A cubic; invariant in pool size
          exactly, and in $(\lambda,m,\delta)$ below the objective's flatness.
    \item So both proposals I opened with are right in direction, and about
          twice too optimistic.
  \end{itemize}
  \begin{takeaway}
    The rate splits into a channel the block schedule governs and one it cannot
    touch. The floor is genuine but distant: what binds at deployed block times
    is the cost of a block, and the lever against the floor is $\gamma/\delta$.
  \end{takeaway}
  {\footnotesize This is an LP-side input to a larger welfare problem, not a
  prescription --- limitations on the backup slides. Extended version, with all
  proofs: \weblink{https://arxiv.org/abs/2608.30321}{arXiv:2608.30321}.}
\end{frame}

\section{Backup}

\begin{frame}{Limitations}
  \begin{itemize}
    \item \textbf{Objective is LP-LVR-only.} A full welfare treatment must also
          weigh MEV redistribution, finality, censorship resistance and oracle
          freshness --- all of which could push $\dt^{\mathrm{opt}}$ back up.
    \item \textbf{Consensus cost.} Issuance is a chosen compensation policy, not
          a measured resource cost; it overstates where it subsidises growth and
          understates where MEV and fees fund validators.
    \item \textbf{Full-range pool.} Most LVR is paid on concentrated liquidity,
          where the two effects on $G$ run in opposite directions --- so the
          full-range floor is not simply a lower bound there.
    \item \textbf{Idealised arbitrageur:} unlimited capital, zero latency,
          continuous sight of $S_t$ --- weakest exactly in the sub-second regime
          the floor argument concerns.
    \item \textbf{Poisson blocks,} inherited for tractability. Deterministic
          slots give the same shape with $C = -\zeta(1/2)/\sqrt{2\pi} \approx
          0.583$ in place of $1/\sqrt2$, and move $\dt^{\mathrm{opt}}$ by $5\%$.
  \end{itemize}
\end{frame}

\figureslide[$\ell_0(\dt)$ at the ETH/USDT calibration. The dark band is
constant in $\dt$; only the pale band responds to the block schedule.]%
{The two channels, drawn}{fig_rate_vs_dt}

\begin{frame}{Fees split LVR --- they do not reduce it}
  In the accounting of Milionis et al., arbitrage profit and fee income sum to
  the frictionless LVR of Theorem~1. So $\ell(\dt)$ is the arbitrage half:
  LP loss \emph{net} of the fees LPs get back.
  \begin{itemize}
    \item Diffusion channel: the split is $F(\kappa) : 1-F(\kappa)$, and
          $F(\kappa)\to0$. Shorter blocks \emph{do} return diffusion-LVR to LPs
          as fee income.
    \item Jump channel: the split is $\Psi(\gamma/\delta) : 1-\Psi(\gamma/\delta)$
          --- a function of $\gamma/\delta$ alone, with no $\dt$ in it.
    \item At the calibration $\gamma/\delta = 0.026$, so fees recover
          \hl{$4.1\%$} of jump-LVR. The rest is the floor.
  \end{itemize}
  \begin{takeaway}
    The lever against the floor is the ratio of fee to jump size --- fee-tier
    design --- not the block schedule.
  \end{takeaway}
\end{frame}

      $0.30$ (calm)      & $55$ $(43\%)$      & $148$ $(16\%)$     & $524$ $(4\%)$      & $2{,}027$ $(1\%)$ \\
      $0.816$ (baseline) & $377$ $(92\%)$     & $471$ $(73\%)$     & $847$ $(41\%)$     & $2{,}349$ $(15\%)$ \\
      $1.50$ (stress)    & $1{,}625$ $(98\%)$ & $1{,}719$ $(93\%)$ & $2{,}095$ $(76\%)$ & $3{,}597$ $(44\%)$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item The \emph{level} spans a factor of $65$; the \emph{share} runs from
          $1\%$ to $98\%$ and moves the \emph{other} way in $\lambda$.
    \item The share is what decides whether shortening blocks helps at all.
          At $\lambda = 283$/yr it crosses $50\%$ at $\sigma = 0.55$.
  \end{itemize}
  \begin{takeaway}
    Any policy claim about $12$-second blocks is conditional on the
    volatility-and-jump-intensity regime, not on a single point estimate.
  \end{takeaway}
\end{frame}

\begin{frame}{The reset-to-zero shortcut}
  A common simplification is to assume a trade resets the mispricing $z$ to $0$
  rather than to the band edge $\pm\gamma$ --- i.e.\ to use $\Psi(\kappa)$ in
  place of $P_{\mathrm{trade}} = F(\kappa)$.
  \begin{pitfall}[It is not a small error]
    It understates the diffusion term by $2.7\times$ at $12$~s, $76\times$ at
    $2$~s, and $4{\times}10^7$ at $400$~ms --- because $\Psi$ has a Gaussian
    tail where $F$ decays only algebraically.
  \end{pitfall}
  The persistence of the mispricing (Lemma~1) is what makes the diffusion
  channel decay as $\sqrt{\dt}$ rather than vanish, and so it is also what
  keeps that channel material down into the millisecond range.
\end{frame}
